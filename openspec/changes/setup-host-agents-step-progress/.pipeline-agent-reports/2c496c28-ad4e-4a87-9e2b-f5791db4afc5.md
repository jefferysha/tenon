```ts
// kernel host-agent-install.ts
export function hostAgentsDir(host, { homeDir, env, platform }): string {
  const { env: name, fallback } = CONFIG_DIR[host]       // claude: CLAUDE_CONFIG_DIR/.claude, codex: CODEX_HOME/.codex
  return join(absoluteDirectory(env[name], platform) ?? join(homeDir, fallback), 'agents')
}
export function userHostAgentPath(agentsDir, host, name) { return join(agentsDir, posix.basename(hostAgentPath(host, name))) }
// install: const hostMap = { ...(owned[agentsDir] ?? {}) } ... return { next: { ...owned, [agentsDir]: hostMap }, ... }
// cli agent-host.ts
export const agentsDirOf = (paths, host) => hostAgentsDir(host, { homeDir: paths.homeDir, env: paths.hostEnv ?? {}, platform: process.platform })
```

## 遗留问题 / 提醒

1. `docs/usage/cli-reference.md:751-752` 与 `docs/usage/zh-CN/cli-reference.md:251-252` 仍把安装位置写成 `~/.claude/agents/…`、`~/.codex/agents/…`。这两处不在 Task 16 的 Files 里，没有改；对未设变量的环境仍然准确，但没提跟随 `CLAUDE_CONFIG_DIR` / `CODEX_HOME`。建议主线决定是否顺手补一句。
2. 把 `CODEX_HOME` 加进 vitest 的 `ISOLATED_ENV` 后，`packages/cli/src/loop-run.real.integration.test.ts:331` 的 `findCodexHome()` 读 `process.env.CODEX_HOME` 找真实凭证时会看不到开发机上的非默认 `CODEX_HOME`，只回退 `~/.codex`。该用例默认走 honest skip，CI 也不设 `CODEX_HOME`，所以不影响现有结果；只在「本机 `CODEX_HOME` 非默认且手动跑 real 模式」时才有影响。
3. 我把全量 vitest 的输出写到了 scratchpad 的 `vitest-full.log`，覆盖了该目录里原有的同名旧日志（scratchpad 内文件，不属于仓库）。
4. 开发期按宿主名写下的 `claude` / `codex` 清单组按设计不解释、不清理，会一直留在 `host-owned.json` 里；这是规格与设计的明确选择。
5. 清单分组键是 `join()` 归一化后的目录字符串，不做 `realpath`：同一物理目录经符号链接用两种写法会各记一组（同一文件被两组各记一次），后果仅是卸载其中一组后另一组的条目变成「文件已不在」被静默移出，不会误删。
6. 没有执行 `tenon agent install` / `setup` / `update`，没有读写真实 `$HOME`、`~/.claude`、`~/.codex`、`CLAUDE_CONFIG_DIR` 指向的目录；没有 `git commit` / `add` / `stash` 等操作；没有碰 `.pipeline*`、`test-plan.yaml`、`.tenon/tests/*`、`.claude/worktrees/`、`unify-stage-skill-canvas/`、`fix-pr80-repository-hygiene/`；没有被 hook 拦截。
